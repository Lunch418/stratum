# Попадание точки в объект (GetObjectFromPoint2d): линия, её окрестность,
# замкнутая ломаная без заливки и с заливкой, габарит, группа, Z-порядок.
@ fixtures/PROJECTS/samples/BALLS WorkSpace
% Wait = 1
! hs := OpenSchemeWindow("W1", "WorkSpace", "")
! hp := CreatePen2d(~hs, 0, 1, Rgb(0,0,0), 13)
! hb := CreateBrush2d(~hs, 0, 0, Rgb(255,0,0), #0, 13)
# отрезок (100,100)-(200,100)
! ln := CreatePolyLine2d(~hs, ~hp, #0, 100, 100, 200, 100)
# замкнутый треугольник без заливки
! tr := CreatePolyLine2d(~hs, ~hp, #0, 300, 100, 400, 100, 350, 200, 300, 100)
# такой же с заливкой
! tf := CreatePolyLine2d(~hs, ~hp, ~hb, 500, 100, 600, 100, 550, 200, 500, 100)
# два пересекающихся отрезка в группе
! g1 := CreatePolyLine2d(~hs, ~hp, #0, 100, 300, 200, 300)
! g2 := CreatePolyLine2d(~hs, ~hp, #0, 150, 250, 150, 350)
! gg := CreateGroup2d(~hs, ~g1, ~g2)
# два перекрывающихся залитых квадрата: верхний создан позже
! q1 := CreatePolyLine2d(~hs, ~hp, ~hb, 300, 300, 400, 300, 400, 400, 300, 400, 300, 300)
! q2 := CreatePolyLine2d(~hs, ~hp, ~hb, 350, 350, 450, 350, 450, 450, 350, 450, 350, 350)
ln | Float(ln)
