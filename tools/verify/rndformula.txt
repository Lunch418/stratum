# rnd(x): сколько из 3000 значений совпадают с x*r/32767 и x*r*(1/32767)
> i := 0
> while (~i < 3000)
>  v := rnd(8)
>  k := round(~v*32767/8, 0)
>  cO := ~cO + (~v == 8*~k/32767)
>  cA := ~cA + (~v == 8*~k*(1/32767))
>  cB := ~cB + (~v == ~k*(8/32767))
>  w := rnd(3)
>  m := round(~w*32767/3, 0)
>  eO := ~eO + (~w == 3*~m/32767)
>  eA := ~eA + (~w == 3*~m*(1/32767))
>  eB := ~eB + (~w == ~m*(3/32767))
>  i := ~i + 1
> endwhile
