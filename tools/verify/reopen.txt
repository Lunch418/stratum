! hw := CreateWindowEx("W2", "", "WorkSpace", 0, 0, 100, 100, "")
! hp2 := CreatePen2d(~hw, 0, 1, Rgb(0,0,0), 13)
! hl2 := CreatePolyLine2d(~hw, ~hp2, #0, 0, 0, 10, 10)
! h4 := CreateWindowEx("W2", "", "WorkSpace", 0, 0, 100, 100, "")
! h5 := OpenSchemeWindow("W2", "WorkSpace", "")

same3 | Float(h3) - Float(hs)
same4 | Float(h4) - Float(hw)
same5 | Float(h5) - Float(hw)
